<script>
const canvas = document.getElementById('canvas');
const ctx = canvas.getContext('2d');

// --- GAME STATE ---
const bear = {
  screenX: canvas.width / 2,
  screenY: 480, // Fixed lower center position
  scale: 1,
  jumpOffset: 0,
  isJumping: false,
  jumpProgress: 0
};

// Log sequence: X position relative to center, Y position down the trail
let logs = [
  { x: 0,    y: 480 },
  { x: -30,  y: 330 },
  { x: 40,   y: 180 },
  { x: -20,  y: 30 },
  { x: 20,   y: -120 }
];

let targetLogIndex = 0;

// --- WORLD / LEAF MAP TRANSFORM HELPERS ---
// Move entire world (logs map) downward by stepDistance pixels.
function moveWorldDown(stepDistance) {
  logs.forEach(log => {
    log.y += stepDistance;
  });
}

// Find the next target log ahead of the bear: largest y strictly less than bearLegsY.
function findNextTargetLog(bearLegsY) {
  let target = null;
  let maxY = -Infinity;

  logs.forEach(log => {
    if (log.y < bearLegsY && log.y > maxY) {
      maxY = log.y;
      target = log;
    }
  });

  return target;
}

// Lock the world so that the next target log aligns exactly with the bear's legs.
function lockNextLogToBear(bearLegsY) {
  const targetLog = findNextTargetLog(bearLegsY);
  if (!targetLog) return;

  const offsetY = bearLegsY - targetLog.y;

  logs.forEach(log => {
    log.y += offsetY;
  });
}

// --- INPUT HANDLER ---
window.addEventListener('keydown', (e) => {
  if (e.code !== 'Space') return;

  // Do not start a new jump while one is already in progress
  if (bear.isJumping) return;

  // Find the next log ahead of the bear (largest y that is still above bear.screenY)
  const nextLog = findNextTargetLog(bear.screenY);

  // If there is no log ahead, there is nowhere to jump
  if (!nextLog) return;

  // Start jump animation; world will be shifted on landing
  bear.isJumping = true;
  bear.jumpProgress = 0;
});

// --- RENDER & ANIMATION LOOP ---
function update() {
  if (bear.isJumping) {
    // Progress through jump (0.0 to 1.0)
    bear.jumpProgress += 0.03;

    // 1. Arc height (jump upward illusion)
    bear.jumpOffset = Math.sin(bear.jumpProgress * Math.PI) * 40;

    // 2. Scale character to feel like it's leaping closer to camera
    bear.scale = 1 + Math.sin(bear.jumpProgress * Math.PI) * 0.25;

    // End jump landing and perform world target-lock
    if (bear.jumpProgress >= 1) {
      bear.isJumping = false;
      bear.jumpProgress = 0;
      bear.jumpOffset = 0;
      bear.scale = 1;

      // On landing, shift the entire log map downward so that the next
      // target log aligns exactly with the bear's fixed leg position.
      lockNextLogToBear(bear.screenY);
    }
  }
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // --- 1. DRAW LOGS / PILLARS ---
  logs.forEach((log) => {
    // Depth scaling: smaller near the top of screen, full size at bottom
    const depthScale = Math.max(0.4, Math.min(1.0, log.y / canvas.height));
    const width = 80 * depthScale;
    const height = 40 * depthScale;

    ctx.save();
    ctx.translate(canvas.width / 2 + log.x, log.y);

    // Pillar base shadow
    ctx.fillStyle = 'rgba(0,0,0,0.4)';
    ctx.beginPath();
    ctx.ellipse(0, 10 * depthScale, width / 2, height / 3, 0, 0, Math.PI * 2);
    ctx.fill();

    // Wood top surface (Simulated 2D Log)
    ctx.fillStyle = '#795548';
    ctx.beginPath();
    ctx.ellipse(0, 0, width / 2, height / 2, 0, 0, Math.PI * 2);
    ctx.fill();

    // Wood rim border
    ctx.strokeStyle = '#3e2723';
    ctx.lineWidth = 3 * depthScale;
    ctx.stroke();

    ctx.restore();
  });

  // --- 2. DRAW BEAR (FIXED SCREEN POSITION) ---
  ctx.save();
  // Translate to bear's locked camera coordinate (minus jump offset)
  ctx.translate(bear.screenX, bear.screenY - bear.jumpOffset);
  ctx.scale(bear.scale, bear.scale);

  // Bear Shadow on log
  ctx.fillStyle = 'rgba(0,0,0,0.3)';
  ctx.beginPath();
  ctx.ellipse(0, bear.jumpOffset + 10, 20, 8, 0, 0, Math.PI * 2);
  ctx.fill();

  // Draw Bear (Placeholder back-view vector sprite)
  // [Replace this block with ctx.drawImage(bearImage, ...)]
  ctx.fillStyle = '#d35400'; // Orange bear fur color
  ctx.beginPath();
  ctx.arc(0, -20, 22, 0, Math.PI * 2); // Body
  ctx.arc(-15, -38, 7, 0, Math.PI * 2); // Left Ear
  ctx.arc(15, -38, 7, 0, Math.PI * 2);  // Right Ear
  ctx.fill();

  ctx.restore();
}

function loop() {
  update();
  draw();
  requestAnimationFrame(loop);
}

loop();
</script>
